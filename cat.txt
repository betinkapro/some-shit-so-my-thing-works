local function new(class, props, children)
	local inst = Instance.new(class)
	local parent
	for k, v in pairs(props or {}) do
		if k == "Parent" then
			parent = v
		else
			inst[k] = v
		end
	end
	for _, child in ipairs(children or {}) do
		child.Parent = inst
	end
	inst.Parent = parent
	return inst
end

local function corner(inst, radius)
	return new("UICorner", { CornerRadius = UDim.new(0, radius or 8), Parent = inst })
end

local function stroke(inst, color, thickness)
	return new("UIStroke", {
		Color = color or Theme.Stroke,
		Thickness = thickness or 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		Parent = inst,
	})
end

local function text(parent, str, size, color, font, props)
	local label = new("TextLabel", {
		BackgroundTransparency = 1,
		Text = str,
		TextSize = size or 14,
		TextColor3 = color or Theme.Text,
		Font = font or Theme.Font,
		TextXAlignment = Enum.TextXAlignment.Left,
		Parent = parent,
	})
	for k, v in pairs(props or {}) do
		label[k] = v
	end
	return label
end

local function createPenguin(parent)
	local root = new("Frame", {
		Name = "Penguin",
		Size = UDim2.fromOffset(26, 26),
		BackgroundTransparency = 1,
		Parent = parent,
	})
	local function part(color, x, y, w, h)
		local f = new("Frame", {
			Position = UDim2.fromOffset(x, y),
			Size = UDim2.fromOffset(w, h),
			BackgroundColor3 = color,
			BorderSizePixel = 0,
			Parent = root,
		})
		new("UICorner", { CornerRadius = UDim.new(0.5, 0), Parent = f })
		return f
	end
	local orange = Color3.fromRGB(255, 176, 36)
	local black = Color3.fromRGB(28, 28, 34)
	local white = Color3.fromRGB(245, 245, 250)
	part(orange, 2, 22, 9, 4)
	part(orange, 15, 22, 9, 4)

local function recolorAccent(old, new)
	local props = { "BackgroundColor3", "TextColor3", "ImageColor3", "ScrollBarImageColor3" }

	local function sweep(root)
		if not root then return end
		for _, d in ipairs(root:GetDescendants()) do
			for _, p in ipairs(props) do
				local ok, val = pcall(function() return d[p] end)
				if ok and typeof(val) == "Color3" and val == old then
					pcall(function() d[p] = new end)
				end
			end
			if d:IsA("UIStroke") then
				local ok, val = pcall(function() return d.Color end)
				if ok and typeof(val) == "Color3" and val == old then
					pcall(function() d.Color = new end)
				end
			end
		end
	end

	sweep(Library["items"])
	sweep(Library["other"])
end

local PNG_SIG = string.char(137, 80, 78, 71, 13, 10, 26, 10)

local function httpGet(url)
	local ok, body = pcall(function() return game:HttpGet(url) end)
	if ok and type(body) == "string" and #body > 0 then return body, "HttpGet" end

	if syn and syn.request then
		local ok2, r = pcall(syn.request, { Url = url, Method = "GET" })
		if ok2 and r and r.StatusCode == 200 and type(r.Body) == "string" then
			return r.Body, "syn.request"
		end
	end
	if request then
		local ok2, r = pcall(request, { Url = url, Method = "GET" })
		if ok2 and r and r.StatusCode == 200 and type(r.Body) == "string" then
			return r.Body, "request"
		end
	end
	if http and http.request then
		local ok2, r = pcall(http.request, { Url = url, Method = "GET" })
		if ok2 and r and r.StatusCode == 200 and type(r.Body) == "string" then
			return r.Body, "http.request"
		end
	end
end

local function pngSize(body)
	if #body < 24 or body:sub(1, 8) ~= PNG_SIG then return nil end
	local w = body:byte(17) * 16777216 + body:byte(18) * 65536
			+ body:byte(19) * 256 + body:byte(20)
	local h = body:byte(21) * 16777216 + body:byte(22) * 65536
			+ body:byte(23) * 256 + body:byte(24)
	if w <= 0 or h <= 0 or w > 20000 or h > 20000 then return nil end
	return w, h
end

local function contentTag(s)
	local h = 5381
	for i = 1, #s do h = (h * 33 + s:byte(i)) % 2147483647 end
	return string.format("%08x", h)
end

local imgW, imgH

local function downloadBackground()
	local body, via = httpGet(ASSET.url)
	if not body then return nil, "HTTP failed (no HttpGet/request)" end
	log("HTTP ok via %s, %d B", tostring(via), #body)

	if body:sub(1, 8) ~= PNG_SIG then
		return nil, ("not a PNG (starts with: %s)"):format(body:sub(1, 60):gsub("[%c]", "?"))
	end

	local w, h = pngSize(body)
	if w then
		imgW, imgH = w, h
		log("image size: %dx%d (aspect %.3f)", w, h, w / h)
	else
		log("could not read PNG dimensions - 16:9 fallback")
		imgW, imgH = 16, 9
	end

	for _, f in ipairs({ ASSET.dir, "kawaii.gg" }) do pcall(makefolder, f) end

	local path = ("%s/bg_%s.png"):format(ASSET.dir, contentTag(body))
	if not ASSET.force_redownload and isfile(path) then
		log("disk cache: %s", path)
		return path
	end

	local okW, err = pcall(writefile, path, body)
	if not okW then return nil, "writefile failed: " .. tostring(err) end
	log("wrote: %s (%d B)", path, #body)
	return path
end

local function resolveCustom(path)
	for _ = 1, 10 do
		local ok, asset = pcall(getcustomasset, path)
		if ok and type(asset) == "string" and #asset > 0 then return asset end
		task.wait(0.2)
	end
end

local function inBackground(d)
	local p = d
	while p do
		if p.Name == "KawaiiBackgroundHolder" or p.Name == "KawaiiScreenLayer" then
			return true
		end
		p = p.Parent
	end
	return false
end

local function applyPanels()
	for _, d in ipairs(win:GetDescendants()) do
		if (d:IsA("Frame") or d:IsA("ImageButton") or d:IsA("TextButton"))
			and not inBackground(d) then
			local size = d.AbsoluteSize
			if size.X > 0 and size.Y > 0 then
				local area = size.X * size.Y
				if area >= ASSET.panel_area then
					if d.BackgroundTransparency < ASSET.panel_alpha then
						d.BackgroundTransparency = ASSET.panel_alpha
					end
				else
					if d.BackgroundTransparency < ASSET.button_alpha then
						d.BackgroundTransparency = ASSET.button_alpha
					end
				end
			end
		end
	end
end

local function keepPanelsTransparent()
	applyPanels()
	local t0 = os.clock()
	local conn
	conn = track(RunService.Heartbeat:Connect(function()
		if os.clock() - t0 > 20 then conn:Disconnect() return end
		applyPanels()
	end))
	track(win.DescendantAdded:Connect(function()
		task.delay(0.3, applyPanels)
	end))
end

local bgLabel, bgHolder

local function layoutImage()
	if not bgLabel then return end

	local W = win.AbsoluteSize.X * ASSET.coverage
	local H = win.AbsoluteSize.Y * ASSET.coverage
	if W <= 1 or H <= 1 then return end

	if ASSET.fit == "cover" then
		bgLabel.Size        = UDim2.fromScale(ASSET.coverage, ASSET.coverage)
		bgLabel.Position    = UDim2.fromScale(0.5, ASSET.anchor_y)
		bgLabel.AnchorPoint = Vector2.new(0.5, ASSET.anchor_y)
		bgLabel.ScaleType   = Enum.ScaleType.Crop
		return
	end

	local aspect = imgW / imgH
	local dispW, dispH
	if W / H > aspect then
		dispH = H
		dispW = H * aspect
	else
		dispW = W
		dispH = W / aspect
	end

	bgLabel.ScaleType   = Enum.ScaleType.Fit
	bgLabel.AnchorPoint = Vector2.new(0.5, ASSET.anchor_y)
	bgLabel.Position    = UDim2.fromScale(0.5, ASSET.anchor_y)
	bgLabel.Size        = UDim2.fromOffset(math.floor(dispW), math.floor(dispH))
end

local menuOpen = true

local function setMenu(open)
	open = not not open
	if open == menuOpen then return end
	menuOpen = open

	itemsGui.Enabled = open
	if win then win.Visible = open end

	if toggleButton then
		toggleButton.BackgroundColor3 = open
			and ACCENT
			or  Color3.fromRGB(70, 70, 78)
		toggleButton.Text = open and "kawaii.gg" or "kawaii.gg (off)"
	end
end

local function toggleMenu() setMenu(not menuOpen) end

local rawToggle = window.toggle_menu

	if bool == nil then toggleMenu() else setMenu(bool) end
end

local Combat = (function()
	local M = {}

	function M.Click()
		if typeof(mouse1click) == "function" then
			if pcall(mouse1click) then return true end
		end
		if typeof(mouse1press) == "function" and typeof(mouse1release) == "function" then
			local ok = pcall(function()
				mouse1press()
				task.wait()
				mouse1release()
			end)
			if ok then return true end
		end
		if not M._warned then
			M._warned = true
			warn("[kawaii.gg] This executor exposes no mouse1click/mouse1press, auto-fire can't simulate a click")
		end
		return false
	end

	function M.Click2()
		if typeof(mouse2click) == "function" then
			if pcall(mouse2click) then return true end
		end
		if typeof(mouse2press) == "function" and typeof(mouse2release) == "function" then
			local ok = pcall(function()
				mouse2press()
				task.wait()
				mouse2release()
			end)
			if ok then return true end
		end
		if not M._warned2 then
			M._warned2 = true
			warn("[kawaii.gg] This executor exposes no mouse2click/mouse2press, right-click spam can't simulate a click")
		end
		return false
	end

	local function aliveRoot(plr)
		local char = plr.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and char:FindFirstChild("HumanoidRootPart")
		if not hum or not root or hum.Health <= 0 then return nil end
		if not (root.Position.Magnitude < 1e9) then return nil end
		return root
	end
	M.AliveRoot = aliveRoot

	local function enemies(teamCheck)
		local list = {}
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= player and not (teamCheck and plr.Team ~= nil and plr.Team == player.Team) then
				local root = aliveRoot(plr)
				if root then table.insert(list, { player = plr, root = root }) end
			end
		end
		return list
	end
	M.Enemies = enemies

	local visParams = RaycastParams.new()
	visParams.FilterType = Enum.RaycastFilterType.Exclude
	visParams.RespectCanCollide = true

	function M.IsVisible(targetRoot, fromPos)
		local char = getCharacter()
		visParams.FilterDescendantsInstances = { char, targetRoot.Parent }
		local hit = workspace:Raycast(fromPos, targetRoot.Position - fromPos, visParams)
		return hit == nil
	end

	local crossParams = RaycastParams.new()
	crossParams.FilterType = Enum.RaycastFilterType.Exclude
	crossParams.RespectCanCollide = true

	function M.CrosshairTarget(maxDist, teamCheck)
		local cam = workspace.CurrentCamera
		local char = getCharacter()
		if not cam or not char then return nil end
		crossParams.FilterDescendantsInstances = { char }
		local hit = workspace:Raycast(cam.CFrame.Position, cam.CFrame.LookVector * (maxDist or 1000), crossParams)
		if not hit then return nil end
		local model = hit.Instance:FindFirstAncestorOfClass("Model")
		local plr = model and Players:GetPlayerFromCharacter(model)
		if not plr or plr == player then return nil end
		if teamCheck and plr.Team ~= nil and plr.Team == player.Team then return nil end
		local root = aliveRoot(plr)
		if not root then return nil end
		return plr, root
	end

	function M.FovTarget(fovDeg, teamCheck, visibleOnly)
		local cam = workspace.CurrentCamera
		if not cam then return nil end
		local look, camPos = cam.CFrame.LookVector, cam.CFrame.Position
		local cosLimit = math.cos(math.rad(fovDeg))
		local best, bestDot
		for _, e in ipairs(enemies(teamCheck)) do
			local dot = look:Dot((e.root.Position - camPos).Unit)
			if dot >= cosLimit and (not visibleOnly or M.IsVisible(e.root, camPos)) then
				if not bestDot or dot > bestDot then best, bestDot = e, dot end
			end
		end
		return best
	end

	return M
end)()

local MAX_VOID_DIST = 100000
local VoidGate = { phase = "home", t0 = 0, base = nil }
local voidGateConn = nil

local function voidGateSuspends()
	return State.Ragebot and State.Voidspam and VoidGate.phase == "spam"
end

local function voidGateStep()
	if not State.Ragebot or not State.Voidspam then return end
	local now = os.clock()
	if VoidGate.phase == "spam" then
		if now - VoidGate.t0 >= State.VoidTime then
			VoidGate.phase = "home"
			VoidGate.t0 = now
			VoidGate.base = nil
			return
		end
		local char = getCharacter()
		local root, hum = getRoot(char), getHumanoid(char)
		if not root or not hum or hum.Health <= 0 then return end
		if not VoidGate.base then VoidGate.base = root.Position end
		local fh = workspace.FallenPartsDestroyHeight
		if fh ~= fh then fh = -500 end
		local y = math.max(VoidGate.base.Y, fh + 25)
		local ang = math.random() * math.pi * 2
		local dist = math.min(State.VoidDistance, MAX_VOID_DIST) * (0.7 + math.random() * 0.3)
		root.CFrame = CFrame.new(VoidGate.base.X + math.cos(ang) * dist, y, VoidGate.base.Z + math.sin(ang) * dist)
		root.AssemblyLinearVelocity = Vector3.zero
		root.AssemblyAngularVelocity = Vector3.zero
	else
		if now - VoidGate.t0 >= State.VoidHomeTime then
			VoidGate.phase = "spam"
			VoidGate.t0 = now
			VoidGate.base = nil
		end
	end
end

local function startVoidGate()
	if voidGateConn then return end
	VoidGate.phase = "spam"
	VoidGate.t0 = os.clock()
	VoidGate.base = nil
	voidGateConn = RunService.Heartbeat:Connect(voidGateStep)

	VoidGate.phase = "home"
	VoidGate.base = nil
end

local undergroundActive = false

local function startUnderground()
	if undergroundActive then return end
	clearConnections()
	local char = getCharacter()
	local root = getRoot(char)
	local hum = getHumanoid(char)
	if not root or not hum then return end
	claimOwnership(root)

	if originalHipHeight == 2 then originalHipHeight = hum.HipHeight end
	if originalRootSize == Vector3.new(2, 2, 1) then originalRootSize = root.Size end

	undergroundActive = true
	lastUndergroundPos = root.Position
	direction = 1
	root.CanCollide = false
	for _, part in ipairs(char:GetDescendants()) do
		if part:IsA("BasePart") then part.CanCollide = false end
	end

	local anchor = lastUndergroundPos
	local side = 1
	local acc = 0
	local function step(dt)
		if not State.Ragebot or State.Method ~= "Underground" then return end
		if voidGateSuspends() then return end
		char = getCharacter()
		root = getRoot(char)
		hum = getHumanoid(char)
		if not root or not hum or hum.Health <= 0 then return end

		local fh = workspace.FallenPartsDestroyHeight
		if fh ~= fh then fh = -500 end
		local y = math.max(anchor.Y - State.Depth, fh + 25)

		acc = acc + (dt or 0)
		local interval = 1 / math.max(State.UndergroundSpeed, 1)
		if acc >= interval then
			acc = acc % interval
			side = -side
		end
		local pos = Vector3.new(anchor.X + side * State.UndergroundRange, y, anchor.Z)
		root.CFrame = CFrame.new(pos) * CFrame.Angles(0, math.rad(root.Orientation.Y), 0)
		root.AssemblyLinearVelocity = Vector3.zero
		root.AssemblyAngularVelocity = Vector3.zero
		hum.PlatformStand = true
	end
	table.insert(Connections, RunService.Heartbeat:Connect(step))
end

	if not undergroundActive then return end
	undergroundActive = false
	clearConnections()
	local char = getCharacter()
	local root = getRoot(char)
	local hum = getHumanoid(char)
	if root then
		root.CanCollide = true
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then part.CanCollide = true end
		end
		if lastUndergroundPos then
			root.CFrame = CFrame.new(lastUndergroundPos.X, lastUndergroundPos.Y + 2, lastUndergroundPos.Z)
		end
	end
	if hum then
		hum.PlatformStand = State.HitboxSmaller
		if not State.HitboxSmaller then hum.HipHeight = originalHipHeight end
	end
end

local gunExceptions = { Sniper = true, Crossbow = true, Bow = true, RPG = true }
local gunFields = { ShootSpread = 0, ShootAccuracy = 0, ShootRecoil = 0, ShootCooldown = 0.001, ShootBurstCooldown = 0.001 }
local meleeFields = { AttackCooldown = 0.001, SwingCooldown = 0.001, MeleeCooldown = 0.001, Cooldown = 0.001, RecoveryTime = 0.001, ResetTime = 0.001 }
local fastShootSaved = {}

local function patchFields(data, fields, enabled)
	local saved = fastShootSaved[data]
	if not saved then
		saved = {}
		fastShootSaved[data] = saved
	end
	for field, fast in pairs(fields) do
		if enabled then
			local cur = data[field]
			if typeof(cur) == "number" then
				if saved[field] == nil then saved[field] = cur end
				data[field] = fast
			end
		elseif saved[field] ~= nil then
			data[field] = saved[field]
			saved[field] = nil
		end
	end
end

local function applyFastShoot(enabled)
	if game.GameId ~= 6035872082 then
		if enabled then warn("[kawaii.gg] Fast Shoot is tuned for Rivals (GameId 6035872082)") end
		return
	end
	local ok, Items = pcall(function()
		return require(ReplicatedStorage.Modules.ItemLibrary).Items
	end)
	if not ok or not Items then
		warn("[kawaii.gg] Could not reach ItemLibrary")
		return
	end
	for name, data in pairs(Items) do
		if typeof(data) == "table" then
			if not gunExceptions[name] then pcall(patchFields, data, gunFields, enabled) end
			pcall(patchFields, data, meleeFields, enabled)
		end
	end
end

local Classic = { active = false }

local function classicRoot(plr)
	if plr == player then return nil end
	if State.ClassicTeamCheck and plr.Team ~= nil and plr.Team == player.Team then return nil end
	local char = plr.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not hum or not root or hum.Health <= 0 then return nil end
	if not (root.Position.Magnitude < 1e9) then return nil end
	return root
end

local function classicPick(myRoot)
	local best, bestDist
	for _, plr in ipairs(Players:GetPlayers()) do
		local r = classicRoot(plr)
		if r then
			local d = (r.Position - myRoot.Position).Magnitude
			if not bestDist or d < bestDist then
				best, bestDist = plr, d
			end
		end
	end
	return best
end

local function classicStep()
	if not State.Ragebot or State.Method ~= "Classic" then return end
	if voidGateSuspends() then return end
	local char = getCharacter()
	local root, hum = getRoot(char), getHumanoid(char)
	if not root or not hum or hum.Health <= 0 then return end
	local now0 = os.clock()
	if now0 - (Classic.lastClaim or 0) >= 0.5 then
		Classic.lastClaim = now0
		claimOwnership(root)
	end

	local tRoot = Classic.target and classicRoot(Classic.target)
	if not tRoot then
		local newTarget = classicPick(root)
		if newTarget ~= Classic.target then
			Classic.orbitAngle, Classic.lastOrbitT, Classic.lastClick2 = 0, nil, 0
		end
		Classic.target = newTarget
		tRoot = Classic.target and classicRoot(Classic.target)
		if not tRoot then return end
	end

	if State.ClassicMode == "Melee Orbit" then
		local now = os.clock()
		local dt = now - (Classic.lastOrbitT or now)
		Classic.lastOrbitT = now
		Classic.orbitAngle = (Classic.orbitAngle or 0) + math.rad(State.ClassicOrbitSpeed) * dt
		local offset = Vector3.new(math.cos(Classic.orbitAngle), 0, math.sin(Classic.orbitAngle)) * 1
		local pos = tRoot.Position + offset
		root.CFrame = CFrame.lookAt(pos, tRoot.Position)
	elseif State.ClassicMode == "Knife Backstab" then
		root.CFrame = tRoot.CFrame * CFrame.new(0, 0, 1)
		local now = os.clock()
		if now - (Classic.lastClick2 or 0) >= State.ClassicBackstabRate then
			Combat.Click2()
			Classic.lastClick2 = now
		end
	else
		local cf = tRoot.CFrame
		if State.ClassicOffset > 0 then cf = cf * CFrame.new(0, 0, State.ClassicOffset) end
		root.CFrame = cf
	end
	root.AssemblyLinearVelocity = Vector3.zero
	root.AssemblyAngularVelocity = Vector3.zero
end

local function startClassic()
	if Classic.active then return end
	clearConnections()
	local char = getCharacter()
	local root = getRoot(char)
	if not root then return end
	claimOwnership(root)
	Classic.active = true
	Classic.char = char
	Classic.home = root.CFrame
	Classic.target = nil
	Classic.orbitAngle, Classic.lastOrbitT, Classic.lastClick2 = 0, nil, 0
	for _, part in ipairs(char:GetDescendants()) do
		if part:IsA("BasePart") then part.CanCollide = false end
	end
	for _, signal in ipairs({ RunService.Stepped, RunService.Heartbeat, RunService.RenderStepped }) do
		table.insert(Connections, signal:Connect(classicStep))
	end
end

local function stopClassic()
	if not Classic.active then return end
	Classic.active = false
	clearConnections()
	local char = getCharacter()
	local root = getRoot(char)
	if root and Classic.char == char and Classic.home then
		root.CFrame = Classic.home
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then part.CanCollide = true end
		end
	end
	Classic.target, Classic.home = nil, nil
end

		{ "UpperTorso", "LeftUpperArm" }, { "LeftUpperArm", "LeftLowerArm" }, { "LeftLowerArm", "LeftHand" },
		{ "UpperTorso", "RightUpperArm" }, { "RightUpperArm", "RightLowerArm" }, { "RightLowerArm", "RightHand" },
		{ "LowerTorso", "LeftUpperLeg" }, { "LeftUpperLeg", "LeftLowerLeg" }, { "LeftLowerLeg", "LeftFoot" },
		{ "LowerTorso", "RightUpperLeg" }, { "RightUpperLeg", "RightLowerLeg" }, { "RightLowerLeg", "RightFoot" },
	}
	local R6_EDGES = {
		{ "Head", "Torso" }, { "Torso", "Left Arm" }, { "Torso", "Right Arm" },
		{ "Torso", "Left Leg" }, { "Torso", "Right Leg" },
	}
	local function edgesFor(char)
		return char:FindFirstChild("UpperTorso") and R15_EDGES or R6_EDGES
	end

	local espObjs = {}

	local function newLine(thick)
		local l = Drawing.new("Line")
		l.Thickness = thick or 1.4
		l.Transparency = 1
		l.Visible = false
		return l
	end

	local function ensureEsp(plr)
		local o = espObjs[plr]
		if o then return o end
		o = {
			skeleton = {}, tracer = newLine(1.4),
			healthBg = newLine(3), healthFg = newLine(3),
			nameText = Drawing.new("Text"),
		}
		o.nameText.Size = 14
		o.nameText.Center = true
		o.nameText.Outline = true
		o.nameText.Transparency = 1
		for i = 1, #R15_EDGES do o.skeleton[i] = newLine(1.2) end
		espObjs[plr] = o
		return o
	end

	local function hideEsp(o)
		o.tracer.Visible = false
		o.healthBg.Visible = false
		o.healthFg.Visible = false
		o.nameText.Visible = false
		for _, l in ipairs(o.skeleton) do l.Visible = false end
	end

	local function destroyEsp(plr)
		local o = espObjs[plr]
		if not o then return end
		o.tracer:Remove()
		o.healthBg:Remove()
		o.healthFg:Remove()
		o.nameText:Remove()
		for _, l in ipairs(o.skeleton) do l:Remove() end
		espObjs[plr] = nil
	end

	local function updateEspFor(plr, cam)
		local char = plr.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and char:FindFirstChild("HumanoidRootPart")
		local head = char and char:FindFirstChild("Head")
		local o = ensureEsp(plr)
		if not hum or not root or not head or hum.Health <= 0 then
			hideEsp(o)
			return
		end
		if V.EspTeamCheck and plr.Team ~= nil and plr.Team == player.Team then
			hideEsp(o)
			return
		end
		local dist = (root.Position - cam.CFrame.Position).Magnitude
		if dist > V.EspMaxDist then
			hideEsp(o)
			return
		end

		local color = Theme.Accent
		if V.EspVisibleColor then
			color = Combat.IsVisible(root, cam.CFrame.Position) and Color3.fromRGB(90, 230, 120) or Color3.fromRGB(255, 70, 70)
		end

		if V.EspSkeleton then
			local edges = edgesFor(char)
			local i = 0
			for _, edge in ipairs(edges) do
				i += 1
				local a, b = char:FindFirstChild(edge[1]), char:FindFirstChild(edge[2])
				local line = o.skeleton[i]
				if a and b then
					local p0, v0 = cam:WorldToViewportPoint(a.Position)
					local p1, v1 = cam:WorldToViewportPoint(b.Position)
					if v0 and v1 then
						line.From, line.To, line.Color = Vector2.new(p0.X, p0.Y), Vector2.new(p1.X, p1.Y), color
						line.Visible = true
					else
						line.Visible = false
					end
				else
					line.Visible = false
				end
			end
			for j = i + 1, #o.skeleton do o.skeleton[j].Visible = false end
		else
			for _, l in ipairs(o.skeleton) do l.Visible = false end
		end

		local headPos, headOn = cam:WorldToViewportPoint(head.Position + Vector3.new(0, 0.6, 0))
		local footPos, footOn = cam:WorldToViewportPoint(root.Position - Vector3.new(0, 3, 0))

		if V.EspTracer and headOn then
			local vp = cam.ViewportSize
			o.tracer.From, o.tracer.To, o.tracer.Color = Vector2.new(vp.X / 2, vp.Y), Vector2.new(headPos.X, headPos.Y), color
			o.tracer.Visible = true
		else
			o.tracer.Visible = false
		end

		if V.EspHealth and headOn and footOn then
			local x, yTop, yBot = headPos.X - 14, headPos.Y, footPos.Y
			o.healthBg.From, o.healthBg.To, o.healthBg.Color = Vector2.new(x, yTop), Vector2.new(x, yBot), Color3.fromRGB(20, 20, 20)
			o.healthBg.Visible = true
			local pct = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
			o.healthFg.From, o.healthFg.To = Vector2.new(x, yBot), Vector2.new(x, yBot - (yBot - yTop) * pct)
			o.healthFg.Color = Color3.fromHSV(pct * 0.33, 1, 1)
			o.healthFg.Visible = true
		else
			o.healthBg.Visible = false
			o.healthFg.Visible = false
		end

		if V.EspName and headOn then
			o.nameText.Text = plr.Name .. "  [" .. math.floor(dist) .. "]"
			o.nameText.Position = Vector2.new(headPos.X, headPos.Y - 16)
			o.nameText.Color = color
			o.nameText.Visible = true
		else
			o.nameText.Visible = false
		end
	end

	local function espStep()
		local cam = workspace.CurrentCamera
		if not V.Esp or not cam or not hasDrawing then
			for _, o in pairs(espObjs) do hideEsp(o) end
			return
		end
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= player then updateEspFor(plr, cam) end
		end
	end
	RunService.RenderStepped:Connect(espStep)
	Players.PlayerRemoving:Connect(destroyEsp)

	function M.Update()
		syncAimbot()
		syncFovCircle()
		syncManualHook()
	end

	function M.Shutdown()
		for k, v in pairs(V) do
			if v == true then V[k] = false end
		end
		M.Update()
		for plr in pairs(espObjs) do destroyEsp(plr) end
	end

	return M
end)()

local Visuals = (function()
	local M = {}
	local Lighting = game:GetService("Lighting")

	local COLORS = {
		White = Color3.fromRGB(255, 255, 255),
		Purple = Color3.fromRGB(150, 80, 255),
		Blue = Color3.fromRGB(70, 140, 255),
		Cyan = Color3.fromRGB(70, 230, 230),
		Green = Color3.fromRGB(70, 230, 120),
		Pink = Color3.fromRGB(255, 100, 200),
		Red = Color3.fromRGB(255, 70, 70),
		Orange = Color3.fromRGB(255, 150, 60),
		Yellow = Color3.fromRGB(255, 220, 70),
	}
	M.ColorNames = { "White", "Purple", "Blue", "Cyan", "Green", "Pink", "Red", "Orange", "Yellow" }

	local SKY = {
		Midday = { clock = 13, ambient = Color3.fromRGB(150, 150, 160), brightness = 2.5, stars = 0 },
		Sunset = { clock = 17.8, ambient = Color3.fromRGB(170, 100, 90), brightness = 2, stars = 0 },
		Dawn = { clock = 5.7, ambient = Color3.fromRGB(150, 120, 150), brightness = 1.6, stars = 800 },
		Night = { clock = 0, ambient = Color3.fromRGB(50, 60, 110), brightness = 1, stars = 3000 },
		Space = { clock = 0, ambient = Color3.fromRGB(25, 25, 50), brightness = 0.5, stars = 5000 },
		Neon = { clock = 0, ambient = Color3.fromRGB(130, 50, 210), brightness = 1.2, stars = 4000 },
		Overcast = { clock = 12, ambient = Color3.fromRGB(125, 130, 140), brightness = 1.2, stars = 0 },
	}
	M.SkyNames = { "Midday", "Sunset", "Dawn", "Night", "Space", "Neon", "Overcast" }

	local V = {
		Sky = false, SkyPreset = "Night", SkyIds = "",
		Ambience = false, AmbientColor = "Purple", Brightness = 2, Exposure = 0,
		Rainbow = false, RainbowSpeed = 0.2,
		Fullbright = false, NoShadows = false, ClockLock = false, ClockTime = 14,
		Fog = false, FogColor = "White", FogStart = 0, FogEnd = 300,
		Bloom = false, BloomIntensity = 1, BloomSize = 24, BloomThreshold = 0.8,
		Color = false, Saturation = 0, Contrast = 0, CCBrightness = 0, Tint = "White",
		Rays = false, RaysIntensity = 0.25, RaysSpread = 1,
		Aurora = false, AuroraSpeed = 1, AuroraHeight = 350, AuroraIntensity = 0.8,
		ThirdPerson = false, TPDistance = 10, TPX = 0, TPY = 0,
		Fov = false, FovValue = 90,
	}
	M.State = V

	local LIGHT_PROPS = {
		"Ambient", "OutdoorAmbient", "Brightness", "ClockTime",
		"ExposureCompensation", "FogColor", "FogStart", "FogEnd", "GlobalShadows",
	}
	local orig, touched, lightConn = nil, {}, nil
	local skyObj, skiesHidden = nil, false

	local function snapshot()
		if orig then return end
		orig = { skies = {} }
		for _, p in ipairs(LIGHT_PROPS) do orig[p] = Lighting[p] end
		for _, c in ipairs(Lighting:GetChildren()) do
			if c:IsA("Sky") then table.insert(orig.skies, c) end
		end
	end

	local function target()
		local t = {}
		if V.Sky then
			local p = SKY[V.SkyPreset]
			if p then
				t.ClockTime = p.clock
				t.Ambient = p.ambient
				t.OutdoorAmbient = p.ambient
				t.Brightness = p.brightness
			end
		end
		if V.Ambience then
			local c = COLORS[V.AmbientColor] or COLORS.White
			t.Ambient = c
			t.OutdoorAmbient = c
			t.Brightness = V.Brightness
			t.ExposureCompensation = V.Exposure
		end
		if V.Rainbow then
			local c = Color3.fromHSV((os.clock() * V.RainbowSpeed) % 1, 0.75, 1)
			t.Ambient = c
			t.OutdoorAmbient = c
		end
		if V.Fullbright then
			t.Ambient = Color3.new(1, 1, 1)
			t.OutdoorAmbient = Color3.new(1, 1, 1)
			t.Brightness = 2
			t.ClockTime = 14
			t.FogEnd = 1e6
		end
		if V.Fog then
			t.FogColor = COLORS[V.FogColor] or COLORS.White
			t.FogStart = V.FogStart
			t.FogEnd = math.max(V.FogEnd, V.FogStart + 1)
		end
		if V.ClockLock then t.ClockTime = V.ClockTime end
		if V.NoShadows or V.Fullbright then t.GlobalShadows = false end
		return t
	end

	local function step()
		if not orig then return end
		local t = target()
		for k, v in pairs(t) do
			touched[k] = true
			if Lighting[k] ~= v then Lighting[k] = v end
		end
		for k in pairs(touched) do
			if t[k] == nil then
				Lighting[k] = orig[k]
				touched[k] = nil
			end
		end
	end

	local function syncLighting()
		if next(target()) ~= nil then
			snapshot()
			step()
			if not lightConn then lightConn = RunService.Heartbeat:Connect(step) end
		else
			if orig then step() end
			if lightConn then
				lightConn:Disconnect()
				lightConn = nil
			end
			if next(touched) == nil then orig = nil end
		end
	end

	local FACES = { "SkyboxBk", "SkyboxDn", "SkyboxFt", "SkyboxLf", "SkyboxRt", "SkyboxUp" }
	local DEFAULT_FACES = {
		"rbxasset://textures/sky/sky512_bk.tex", "rbxasset://textures/sky/sky512_dn.tex",
		"rbxasset://textures/sky/sky512_ft.tex", "rbxasset://textures/sky/sky512_lf.tex",
		"rbxasset://textures/sky/sky512_rt.tex", "rbxasset://textures/sky/sky512_up.tex",
	}

	local function applyIds()
		local ids = {}
		for token in string.gmatch(V.SkyIds, "%d+") do table.insert(ids, token) end
		for i, face in ipairs(FACES) do
			if #ids == 0 then
				skyObj[face] = DEFAULT_FACES[i]
			else
				skyObj[face] = "rbxassetid://" .. ((#ids >= 6) and ids[i] or ids[1])
			end
		end
	end

	local function syncSky()
		if V.Sky then
			snapshot()
			if not skiesHidden then
				for _, sk in ipairs(orig.skies) do pcall(function() sk.Parent = nil end) end
				skiesHidden = true
			end
			if not skyObj or not skyObj.Parent then
				skyObj = Instance.new("Sky")
				skyObj.Name = "Kawaii_Sky"
				skyObj.Parent = Lighting
			end
			local p = SKY[V.SkyPreset]
			skyObj.StarCount = p and p.stars or 3000
			applyIds()
		else
			if skyObj then
				skyObj:Destroy()
				skyObj = nil
			end
			if skiesHidden and orig then
				for _, sk in ipairs(orig.skies) do pcall(function() sk.Parent = Lighting end) end
			end
			skiesHidden = false
		end
	end

	local fx = {}
	local function effect(class, name, on, props)
		local inst = fx[name]
		if on then
			if not inst or not inst.Parent then
				inst = Instance.new(class)
				inst.Name = "Kawaii_" .. name
				inst.Parent = Lighting
				fx[name] = inst
			end
			for k, v in pairs(props) do inst[k] = v end
		elseif inst then
			inst:Destroy()
			fx[name] = nil
		end
	end

	local aurora = { folder = nil, conn = nil, ribbons = nil, lastIntensity = nil }

	local function stopAurora()
		if aurora.conn then
			aurora.conn:Disconnect()
			aurora.conn = nil
		end
		if aurora.folder then
			aurora.folder:Destroy()
			aurora.folder = nil
		end
		aurora.ribbons = nil
	end

	local function auroraTransparency(intensity, extra)
		local mid = math.min(1, 1 - 0.85 * intensity + extra)
		local edge = math.min(1, mid + 0.35)
		return NumberSequence.new({
			NumberSequenceKeypoint.new(0, 1),
			NumberSequenceKeypoint.new(0.15, edge),
			NumberSequenceKeypoint.new(0.5, mid),
			NumberSequenceKeypoint.new(0.85, edge),
			NumberSequenceKeypoint.new(1, 1),
		})
	end

	local function startAurora()
		stopAurora()
		local folder = Instance.new("Folder")
		folder.Name = "Kawaii_Aurora"
		folder.Parent = workspace

		local function anchor()
			local part = Instance.new("Part")
			part.Anchored = true
			part.CanCollide = false
			part.CanQuery = false
			part.CanTouch = false
			part.CastShadow = false
			part.Transparency = 1
			part.Size = Vector3.new(1, 1, 1)
			part.Parent = folder
			local att = Instance.new("Attachment")
			att.CFrame = CFrame.Angles(0, 0, math.pi / 2)
			att.Parent = part
			return part, att
		end

		local ribbons = {}
		for i = 1, 4 do
			local p0, a0 = anchor()
			local p1, a1 = anchor()
			local beams = {}
			for j = 1, 2 do
				local beam = Instance.new("Beam")
				beam.Attachment0 = a0
				beam.Attachment1 = a1
				beam.FaceCamera = true
				beam.LightEmission = 1
				beam.LightInfluence = 0
				beam.Segments = 40
				local w = (j == 1) and 170 or 70
				beam.Width0 = w
				beam.Width1 = w
				beam.Parent = p0
				beams[j] = beam
			end
			ribbons[i] = { p0 = p0, p1 = p1, beams = beams, phase = i * 1.9 }
		end
		aurora.folder = folder
		aurora.ribbons = ribbons
		aurora.lastIntensity = nil

		aurora.conn = RunService.Heartbeat:Connect(function()
			local cam = workspace.CurrentCamera
			if not cam or not aurora.ribbons then return end
			local base = cam.CFrame.Position
			local t = os.clock() * V.AuroraSpeed
			local restyle = aurora.lastIntensity ~= V.AuroraIntensity
			aurora.lastIntensity = V.AuroraIntensity
			for i, r in ipairs(aurora.ribbons) do
				local z = base.Z - 650 - i * 110
				local y = V.AuroraHeight + math.sin(t * 0.4 + r.phase) * 35 + i * 20
				r.p0.Position = Vector3.new(base.X - 900, y, z + math.sin(t * 0.3 + r.phase) * 90)
				r.p1.Position = Vector3.new(base.X + 900, y + math.sin(t * 0.5 + r.phase * 2) * 60, z + math.cos(t * 0.3 + r.phase) * 90)
				local hue = 0.38 + 0.1 * math.sin(t * 0.25 + r.phase)
				local seq = ColorSequence.new({
					ColorSequenceKeypoint.new(0, Color3.fromHSV(hue, 0.8, 1)),
					ColorSequenceKeypoint.new(0.5, Color3.fromHSV((hue + 0.18) % 1, 0.85, 1)),
					ColorSequenceKeypoint.new(1, Color3.fromHSV((hue + 0.35) % 1, 0.8, 1)),
				})
				for j, beam in ipairs(r.beams) do
					beam.Color = seq
					beam.CurveSize0 = math.sin(t * 0.8 + r.phase) * 140
					beam.CurveSize1 = math.cos(t * 0.7 + r.phase) * 140
					if restyle then
						beam.Transparency = auroraTransparency(V.AuroraIntensity, j == 1 and 0.4 or 0)
					end
				end
			end
		end)
	end

	local STEP_NAME = "Kawaii_VisualsCamera"
	local renderBound = false
	local origFov = nil
	local tpParams = RaycastParams.new()
	tpParams.FilterType = Enum.RaycastFilterType.Exclude
	tpParams.RespectCanCollide = true
	local charParts, lastScan = {}, 0

	local function onRender()
		local cam = workspace.CurrentCamera
		if not cam then return end
		if V.Fov then
			origFov = origFov or cam.FieldOfView
			cam.FieldOfView = V.FovValue
		end
		if V.ThirdPerson then
			local char = player.Character
			if char then
				if os.clock() - lastScan > 1 then
					lastScan = os.clock()
					charParts = {}
					for _, d in ipairs(char:GetDescendants()) do
						if d:IsA("BasePart") then table.insert(charParts, d) end
					end
				end
				for _, part in ipairs(charParts) do part.LocalTransparencyModifier = 0 end
				tpParams.FilterDescendantsInstances = { char }
			end
			local cf = cam.CFrame
			local look = cf.LookVector
			local goal = cf.Position - look * V.TPDistance + cf.RightVector * V.TPX + cf.UpVector * V.TPY
			local dir = goal - cf.Position
			local hit = workspace:Raycast(cf.Position, dir, tpParams)
			if hit then
				goal = cf.Position + dir.Unit * math.max(hit.Distance - 0.5, 0.5)
			end
			cam.CFrame = CFrame.lookAt(goal, goal + look)
		end
	end

	local function syncRender()
		local need = V.ThirdPerson or V.Fov
		if need and not renderBound then
			renderBound = true
			RunService:BindToRenderStep(STEP_NAME, Enum.RenderPriority.Last.Value, onRender)
		elseif not need and renderBound then
			renderBound = false
			RunService:UnbindFromRenderStep(STEP_NAME)
			local cam = workspace.CurrentCamera
			if cam and origFov then cam.FieldOfView = origFov end
			origFov = nil
		elseif renderBound and not V.Fov and origFov then
			local cam = workspace.CurrentCamera
			if cam then cam.FieldOfView = origFov end
			origFov = nil
		end
	end

	function M.Update()
		syncSky()
		syncLighting()
		effect("BloomEffect", "Bloom", V.Bloom, {
			Intensity = V.BloomIntensity, Size = V.BloomSize, Threshold = V.BloomThreshold,
		})
		effect("ColorCorrectionEffect", "Color", V.Color, {
			Saturation = V.Saturation, Contrast = V.Contrast,
			Brightness = V.CCBrightness, TintColor = COLORS[V.Tint] or COLORS.White,
		})
		effect("SunRaysEffect", "Rays", V.Rays, { Intensity = V.RaysIntensity, Spread = V.RaysSpread })
		if V.Aurora then
			if not aurora.folder then startAurora() end
		else
			stopAurora()
		end
		syncRender()
	end

	function M.Shutdown()
		for k, v in pairs(V) do
			if v == true then V[k] = false end
		end
		M.Update()
	end

	return M
end)()

-- CONFIG (json files in executor workspace, folder kept as LinuxHub
-- so previously saved configs still load)
---------------------------------------------------------------------
local Config = (function()
	local M = {}
	local HttpService = game:GetService("HttpService")
	local ROOT = "LinuxHub"
	local DIR = "LinuxHub/configs"
	M.Dir = DIR

	local function hasFS()
		return typeof(writefile) == "function" and typeof(readfile) == "function"
			and typeof(isfile) == "function" and typeof(isfolder) == "function"
			and typeof(makefolder) == "function" and typeof(listfiles) == "function"
			and typeof(delfile) == "function"
	end
	M.Available = hasFS

	local function ensureDir()
		if not isfolder(ROOT) then makefolder(ROOT) end
		if not isfolder(DIR) then makefolder(DIR) end
	end

	local function clean(name)
		name = tostring(name or "")
		name = string.gsub(name, "[^%w_%- ]", "")
		name = string.gsub(name, "^%s+", "")
		name = string.gsub(name, "%s+$", "")
		return string.sub(name, 1, 32)
	end

	local function path(name)
		return DIR .. "/" .. name .. ".json"
	end

	local function collect()
		local flags = {}
		for key, entry in pairs(Flags) do
			flags[key] = entry.obj:Get()
		end
		return { version = 1, flags = flags }
	end

	local EXPECT = { toggle = "boolean", slider = "number", dropdown = "string", textbox = "string" }

	function M.List()
		local names = {}
		if not hasFS() then return names end
		pcall(ensureDir)
		local ok, files = pcall(listfiles, DIR)
		if ok and type(files) == "table" then
			for _, f in ipairs(files) do
				local n = string.match(tostring(f), "([^/\\]+)%.json$")
				if n then table.insert(names, n) end
			end
		end
		table.sort(names)
		return names
	end

	local function write(name)
		local ok, err = pcall(function()
			ensureDir()
			writefile(path(name), HttpService:JSONEncode(collect()))
		end)
		if ok then return true, name end
		return false, tostring(err)
	end

	function M.Create(name)
		if not hasFS() then return false, "executor has no file api" end
		name = clean(name)
		if name == "" then return false, "enter a name first" end
		ensureDir()
		if isfile(path(name)) then return false, name .. " already exists, use Overwrite" end
		return write(name)
	end

	function M.Overwrite(name)
		if not hasFS() then return false, "executor has no file api" end
		name = clean(name)
		if name == "" or not isfile(path(name)) then return false, "config does not exist yet, use Create" end
		return write(name)
	end

	function M.Delete(name)
		if not hasFS() then return false, "executor has no file api" end
		name = clean(name)
		if name == "" or not isfile(path(name)) then return false, "config does not exist" end
		local ok, err = pcall(delfile, path(name))
		if ok then return true, name end
		return false, tostring(err)
	end

	function M.Load(name)
		if not hasFS() then return false, "executor has no file api" end
		name = clean(name)
		if name == "" or not isfile(path(name)) then return false, "config does not exist" end
		local ok, data = pcall(function()
			return HttpService:JSONDecode(readfile(path(name)))
		end)
		if not ok or type(data) ~= "table" or type(data.flags) ~= "table" then
			return false, "config file is corrupt"
		end
		local applied = 0
		for key, value in pairs(data.flags) do
			local entry = Flags[key]
			if entry and type(value) == EXPECT[entry.kind] then
				if pcall(function() entry.obj:Set(value) end) then applied = applied + 1 end
			end
		end
		return true, applied
	end

	return M
end)()

local function bindFlag(key, kind, getter, setter)
	if not key then return end
	Flags[key] = {
		kind = kind,
		obj = {
			Get = function() return getter() end,
			Set = function(_, v) setter(v) end,
		},
	}
end

-- best-effort visual sync on config load; gameplay always applies even
-- if this build of the library exposes no element setter
local function trySet(el, v)
	for _, m in ipairs({ "set", "Set", "set_value", "update_value" }) do
		if type(el[m]) == "function" then
			local ok = pcall(el[m], el, v)
			if ok then return true end
		end
	end
	return false
end

local function uiToggle(sec, o)
	local value = o.default == true
	local el = sec:toggle({
		name = o.name,
		default = value,
		info = o.info,
		seperator = o.seperator,
		callback = function(state)
			if type(state) ~= "boolean" or state == value then return end
			value = state
			if o.callback then task.spawn(o.callback, state) end
		end,
	})
	if o.flag then
		bindFlag(o.flag, "toggle", function() return value end, function(v)
			value = v
			if o.callback then task.spawn(o.callback, v) end
			trySet(el, v)
		end)
	end
	return el
end

local PRESET_NAMES = { "Purple", "Blue", "Cyan", "Green", "Pink", "Red", "Orange", "Yellow" }

local currentAccent = ACCENT
local function setAccent(c)
	if typeof(c) ~= "Color3" then return end
	pcall(function() Library:update_theme("accent", c) end)
	pcall(function() window:update_theme("accent", c) end)
	recolorAccent(currentAccent, c)
	currentAccent = c
	Theme.Accent = c
end

local rgb = {
	on = false, speed = 0.5, conn = nil, scan = 0,
	orig = setmetatable({}, { __mode = "k" }),
	list = {},
}

local function rgbCollect()
	rgb.list = {}
	if not itemsGui then return end
	for _, d in ipairs(itemsGui:GetDescendants()) do
		if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
			table.insert(rgb.list, d)
			if rgb.orig[d] == nil then rgb.orig[d] = d.TextColor3 end
		end
	end
end

local function rgbStep()
	local now = os.clock()
	if now - rgb.scan > 0.5 then
		rgb.scan = now
		rgbCollect()
	end
	local base = now * rgb.speed
	for _, d in ipairs(rgb.list) do
		if d.Parent then
			local p = d.AbsolutePosition
			d.TextColor3 = Color3.fromHSV((base + p.X / 900 + p.Y / 700) % 1, 0.75, 1)
		end
	end
end

local function setRGB(on)
	if on == rgb.on then return end
	rgb.on = on
	if on then
		rgb.scan = 0
		rgbCollect()
		rgb.conn = RunService.RenderStepped:Connect(rgbStep)
	else
		if rgb.conn then rgb.conn:Disconnect() rgb.conn = nil end
		for d, c in pairs(rgb.orig) do
			if d.Parent then d.TextColor3 = c end
		end
		rgb.orig = setmetatable({}, { __mode = "k" })
		rgb.list = {}
	end
end

	pcall(function()
		State.Ragebot = false
		State.PrimaryFix = false
		if State.FastShoot then
			State.FastShoot = false
			applyFastShoot(false)
		end
		stopMethod()
		if State.HitboxSmaller then
			State.HitboxSmaller = false
			applyHitboxSmaller(false)
		end
		State.FakePosition = false
		stopFakePosition()
		Visuals.Shutdown()
		Legit.Shutdown()
		HUD.Shutdown()
		if rgb.conn then rgb.conn:Disconnect() rgb.conn = nil end
		for _, c in ipairs(uiConns) do pcall(function() c:Disconnect() end) end
		table.clear(uiConns)
		if respawnConn then respawnConn:Disconnect() end
		if itemsGui then itemsGui:Destroy() end
	end)
end
